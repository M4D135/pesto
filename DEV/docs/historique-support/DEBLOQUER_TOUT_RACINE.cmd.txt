@echo off
setlocal EnableExtensions DisableDelayedExpansion
title Pesto - Debloquer les fichiers du dossier
set "Pesto_UNBLOCK_ROOT=%~dp0"
set "Pesto_UNBLOCK_SELF=%~f0"
set "Pesto_PS=%SystemRoot%\System32\WindowsPowerShell\v1.0\powershell.exe"
if exist "%SystemRoot%\Sysnative\WindowsPowerShell\v1.0\powershell.exe" set "Pesto_PS=%SystemRoot%\Sysnative\WindowsPowerShell\v1.0\powershell.exe"
if not exist "%Pesto_PS%" (
    echo ERREUR : Windows PowerShell introuvable.
    pause
    exit /b 1
)
echo.
echo Pesto - DEBLOQUER TOUT
echo Dossier cible : "%Pesto_UNBLOCK_ROOT%"
echo.
echo Retire la marque de telechargement des fichiers de CE dossier uniquement.
echo Aucune modification de l'antivirus, des GPO, des droits ou des verrous Pesto.
echo Continuer seulement si vous faites confiance au contenu de cette archive.
echo.
"%SystemRoot%\System32\choice.exe" /C ON /N /M "Debloquer les fichiers du dossier Pesto ? [O/N] : "
if errorlevel 2 exit /b 2
if errorlevel 1 goto :Pesto_RUN_UNBLOCK
exit /b 2
:Pesto_RUN_UNBLOCK
rem Le code est inclus dans ce CMD : aucun second script bloque n'est necessaire.
rem Le chemin est transmis par l'environnement, jamais insere dans du code.
"%Pesto_PS%" -NoLogo -NoProfile -NonInteractive -Command "$ErrorActionPreference='Stop'; try { $text=[IO.File]::ReadAllText($env:Pesto_UNBLOCK_SELF); $marker='# Pesto_UNBLOCK_POWERSHELL_START'; $pos=$text.LastIndexOf($marker,[StringComparison]::Ordinal); if($pos -lt 0){throw 'Bloc PowerShell absent.'}; & ([ScriptBlock]::Create($text.Substring($pos+$marker.Length))) -Root $env:Pesto_UNBLOCK_ROOT } catch { Write-Host ('ERREUR : '+$_.Exception.Message); exit 1 }"
set "Pesto_EXIT_CODE=%ERRORLEVEL%"
echo.
if "%Pesto_EXIT_CODE%"=="0" echo Termine. Vous pouvez maintenant lancer Pesto.exe.
if not "%Pesto_EXIT_CODE%"=="0" echo Deblocage incomplet. Consulter les erreurs et wad\logs\DEBLOCAGE_Pesto.log si cree.
echo.
pause
exit /b %Pesto_EXIT_CODE%
# Pesto_UNBLOCK_POWERSHELL_START
[CmdletBinding()]
param([Parameter(Mandatory)][string]$Root)
$ErrorActionPreference='Stop'

function Invoke-PestoPackageUnblock {
    [CmdletBinding()]
    param([Parameter(Mandatory)][string]$Root)
    # Aucun lancement de Pesto, aucun changement de politique ou d'ACL.
    $rootItem=Get-Item -LiteralPath $Root -Force -ErrorAction Stop
    if(-not $rootItem.PSIsContainer){throw 'La cible doit etre le dossier Pesto extrait.'}
    if(($rootItem.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0){throw 'Le dossier racine ne doit pas etre un lien ou une jonction.'}
    $trimChars=[char[]]@([char]92,[char]47)
    $full=$rootItem.FullName.TrimEnd($trimChars)
    $volumeRoot=[IO.Path]::GetPathRoot($rootItem.FullName).TrimEnd($trimChars)
    if($full -eq $volumeRoot){throw 'Deblocage refuse a la racine d''un lecteur ou partage.'}
    foreach($required in @('Pesto.exe','wad\Pesto-Startup.ps1','wad\Pesto-Core.ps1')){
        if(-not (Test-Path -LiteralPath (Join-Path $full $required) -PathType Leaf)){
            throw "Dossier Pesto non reconnu : $required absent. Extraire toute l'archive."
        }
    }
    $wadItem=Get-Item -LiteralPath (Join-Path $full 'wad') -Force -ErrorAction Stop
    if(($wadItem.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0){
        throw 'Le dossier wad ne doit pas etre un lien ou une jonction.'
    }
    $prefix=$full+[IO.Path]::DirectorySeparatorChar
    $pending=[Collections.Generic.Stack[string]]::new()
    $pending.Push($full)
    $messages=[Collections.Generic.List[string]]::new()
    $messages.Add('Pesto - deblocage de fichiers - '+[DateTime]::Now.ToString('s'))
    $messages.Add('Dossier : '+$full)
    $scanned=0; $unblocked=0; $alreadyClear=0; $skipped=0; $failures=0
    while($pending.Count -gt 0){
        $directory=$pending.Pop()
        try {
            # Recontroler le dossier avant enumeration, sans suivre les jonctions.
            $dirItem=Get-Item -LiteralPath $directory -Force -ErrorAction Stop
            if(($dirItem.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0){
                $skipped++; $messages.Add('IGNORE LIEN : '+$directory); continue
            }
            $children=@(Get-ChildItem -LiteralPath $directory -Force -ErrorAction Stop)
        } catch {
            $failures++; $messages.Add('ERREUR LECTURE : '+$directory+' | '+$_.Exception.Message); continue
        }
        foreach($entry in $children){
            if(-not $entry.FullName.StartsWith($prefix,[StringComparison]::OrdinalIgnoreCase)){
                $failures++; $messages.Add('ERREUR HORS DOSSIER : '+$entry.FullName); continue
            }
            if(($entry.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0){
                $skipped++; $messages.Add('IGNORE LIEN : '+$entry.FullName); continue
            }
            if($entry.PSIsContainer){$pending.Push($entry.FullName); continue}
            $scanned++
            try {
                # Enumerer les flux : ne pas confondre un acces refuse avec une marque absente.
                $zone=@(Get-Item -LiteralPath $entry.FullName -Stream * -Force -ErrorAction Stop |
                    Where-Object { $_.Stream -eq 'Zone.Identifier' })
                if($zone.Count -eq 0){$alreadyClear++; continue}
                Unblock-File -LiteralPath $entry.FullName -ErrorAction Stop
                $remaining=@(Get-Item -LiteralPath $entry.FullName -Stream * -Force -ErrorAction Stop |
                    Where-Object { $_.Stream -eq 'Zone.Identifier' })
                if($remaining.Count -gt 0){throw 'La marque Zone.Identifier est toujours presente.'}
                $unblocked++; $messages.Add('DEBLOQUE : '+$entry.FullName)
            } catch {
                $failures++; $messages.Add('ERREUR FICHIER : '+$entry.FullName+' | '+$_.Exception.Message)
            }
        }
    }
    $messages.Add(('BILAN : {0} fichiers examines ; {1} debloques ; {2} deja sans marque ; {3} liens ignores ; {4} erreurs.' -f $scanned,$unblocked,$alreadyClear,$skipped,$failures))
    return [pscustomobject]@{
        Root=$full; Scanned=$scanned; Unblocked=$unblocked; AlreadyClear=$alreadyClear
        SkippedLinks=$skipped; Errors=$failures; Success=($failures -eq 0); Messages=$messages.ToArray()
    }
}

$result=Invoke-PestoPackageUnblock -Root $Root
foreach($line in $result.Messages){Write-Host $line}
$logDirectory=Join-Path $result.Root 'wad\logs'
$log=Join-Path $logDirectory 'DEBLOCAGE_Pesto.log'
try {
    # Ne pas rediriger le journal vers un lien ou une jonction.
    foreach($candidate in @((Join-Path $result.Root 'wad'),$logDirectory,$log)){
        if(Test-Path -LiteralPath $candidate){
            $item=Get-Item -LiteralPath $candidate -Force -ErrorAction Stop
            if(($item.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0){
                throw 'Chemin de journal lie refuse.'
            }
        }
    }
    if(-not(Test-Path -LiteralPath $logDirectory -PathType Container)){
        [void][IO.Directory]::CreateDirectory($logDirectory)
    }
    $result.Messages | Set-Content -LiteralPath $log -Encoding UTF8 -ErrorAction Stop
    Write-Host ('Journal : '+$log)
} catch {
    Write-Host ('ERREUR : journal non ecrit : '+$_.Exception.Message)
    exit 1
}
if(-not $result.Success){exit 1}
exit 0
