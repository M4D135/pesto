from pathlib import Path
import hashlib, json, shutil, subprocess, sys, tempfile

ROOT = Path('/Users/david/Documents/Codex/2026-10-26/rep/work/pesto-final-20261028')
WORK = Path(__file__).parent
manifest = json.loads((ROOT/'v2/web/ui-kit.lock.json').read_text())
results = []

def run_case(name, mutate, expected, expected_status, expected_code=None):
    with tempfile.TemporaryDirectory(prefix='ui-kit-fixture-', dir=WORK) as temporary:
        root = Path(temporary)
        files = [entry['path'] for entry in manifest['files']] + [
            'v2/web/ui-kit.lock.json', 'v2/tools/check-ui-kit.py',
            'v2/tools/m4d-components/components.py',
            'v2/tools/m4d-components/components.schema.json',
            'v2/tools/m4d-components/m4d-components.lock.json']
        for relative in files:
            target = root/relative
            target.parent.mkdir(parents=True, exist_ok=True)
            shutil.copyfile(ROOT/relative, target)
        mutate(root)
        process = subprocess.run([sys.executable, '-I', '-S', str(root/'v2/tools/check-ui-kit.py')], capture_output=True, text=True, timeout=15)
        report = json.loads(process.stdout)
        codes = [item['code'] for item in report['findings']]
        assert process.returncode == expected and report['status'] == expected_status, (name, process.returncode, report)
        if expected_code: assert expected_code in codes, (name, codes)
        results.append({'case': name, 'exitCode': process.returncode, 'status': report['status'], 'findingCodes': codes})

def change_json(root, relative, mutate):
    path = root/relative
    value = json.loads(path.read_text())
    mutate(value)
    path.write_text(json.dumps(value))
